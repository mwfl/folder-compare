cmake_minimum_required(VERSION 3.21)
project(folder_compare VERSION 0.1.1 LANGUAGES CXX)

include(FetchContent)
option(FOLDER_COMPARE_USE_LOCAL_MWFL "Use a local mwfl checkout" OFF)
set(FOLDER_COMPARE_MWFL_SOURCE_DIR "../mwfl" CACHE PATH "Local mwfl source directory")
set(MWFL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(FOLDER_COMPARE_USE_LOCAL_MWFL)
    add_subdirectory("${FOLDER_COMPARE_MWFL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/mwfl")
else()
    FetchContent_Declare(mwfl GIT_REPOSITORY https://github.com/mwfl/mwfl.git GIT_TAG v0.1.2)
    FetchContent_MakeAvailable(mwfl)
endif()
if(TARGET mwfl::ui)
    set(FOLDER_COMPARE_MWFL_UI_TARGET mwfl::ui)
else()
    set(FOLDER_COMPARE_MWFL_UI_TARGET mwfl::mwfl)
endif()

add_executable(folder-compare WIN32
    main.cpp compare_model.cpp compare_model.h
    folder_compare.rc resource.h app-icon.ico app.manifest)
target_link_libraries(folder-compare PRIVATE ${FOLDER_COMPARE_MWFL_UI_TARGET} shlwapi)
target_compile_features(folder-compare PRIVATE cxx_std_20)
target_compile_definitions(folder-compare PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(folder-compare PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()

enable_testing()
add_executable(folder-compare-tests compare_model_test.cpp compare_model.cpp compare_model.h)
target_link_libraries(folder-compare-tests PRIVATE ${FOLDER_COMPARE_MWFL_UI_TARGET} shlwapi)
target_compile_features(folder-compare-tests PRIVATE cxx_std_20)
target_compile_definitions(folder-compare-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(folder-compare-tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()
add_test(NAME folder-compare.model COMMAND folder-compare-tests)

install(TARGETS folder-compare RUNTIME DESTINATION . COMPONENT Application)
install(FILES README.md LICENSE DESTINATION . COMPONENT Application)
